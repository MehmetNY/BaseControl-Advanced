# Fan settings: dragging a curve point only makes a draft; the curves reach the laptop only on Apply / Save; boost and the cooling mode follow the same rules.
# Run: ControlCenterAdvanced.exe --sim --script tests/ui/fan_flow.txt --size 1360x1500
say == 1. the page starts in Auto with nothing drafted ==
page fan
wait 600
expect fan.mode auto
let ecBefore snap:fan.cpu.curve
let gpuBefore snap:fan.gpu.curve
dump fan.cpu.curve
expectui fan.d.cpu
expectui fan.d.gpu
shot /tmp/shots/f0_start.png

say == 2. dragging a handle only makes a draft ==
dragby fh#3 0 -50
expectui fan.d.cpu *
expect fan.cpu.curve $ecBefore
expect fan.mode auto
shot /tmp/shots/f1_draft.png

say == 3. discard drops the draft ==
clickid fan.discard
expectui fan.d.cpu

say == 4. Save curves hands the draft to the service but keeps the fan mode ==
dragby fh#3 0 -50
let draft ui:fan.d.cpu
clickid fan.save
wait 2500
expectui fan.d.cpu
expect fan.cpu.curve $draft
expect fan.cpu.curve !$ecBefore
expect fan.gpu.curve $gpuBefore
expect fan.mode auto

say == 5. Apply and use them writes the curves and switches to Custom curve ==
dragby fh#5 0 -30
let draft2 ui:fan.d.cpu
clickid fan.applyuse
wait 2500
expect fan.mode custom
expect fan.cpu.curve $draft2
expectui fan.d.cpu
shot /tmp/shots/f2_custom.png

say == 6. a click on a box selects the point and the arrow keys move it (still a draft) ==
clickid fb#2
key 0x26
key 0x26
key 0x27
expectui fan.d.cpu *
expect fan.cpu.curve $draft2
clickid fan.discard
expectui fan.d.cpu

say == 7. link: one curve for both fans ==
clickid pr#fan.link
expectpref fan.link 1
dragby fh#2 0 -40
let both ui:fan.d.cpu
expectui fan.d.gpu $both
clickid fan.apply
wait 3000
expect fan.cpu.curve $both
expect fan.gpu.curve $both
clickid pr#fan.link
expectpref fan.link 0
shot /tmp/shots/f3_linked.png

say == 8. Turbo fan is an immediate switch that remembers the previous mode ==
clickid tg#fan.boost
wait 1000
expect fan.mode boost
expect fan.boost 1
clickid tg#fan.boost
wait 1000
expect fan.mode custom
expect fan.boost 0

say == 9. the cooling mode needs Confirm ==
clickid cmb#fan.mode
clickid opt#0
expect fan.mode custom
clickid ok#fan.mode
wait 1500
expect fan.mode auto
shot /tmp/shots/f4_auto.png

say == 10. without the vendor's service nothing can be edited or switched ==
external oem.running 0
wait 3000
expect cap.fan.curve !ok
expect cap.fan.ctl !ok
dragby fh#3 0 -50
expectui fan.d.cpu
clickid tg#fan.boost
wait 600
expect fan.mode auto
shot /tmp/shots/f5_offline.png
external oem.running 1
wait 4000
expect oem.link.state connected

say == 11. the vendor's curves: it asks first, Cancel changes nothing, Restore puts the vendor's table back ==
page fan
wait 600
expect fan.cpu.curve !$ecBefore
clickid fan.vendor
wait 500
expectmodal 1
shot /tmp/shots/f6_vendor_ask.png
clickid modal#0
wait 400
expectmodal 0
expect fan.cpu.curve !$ecBefore
clickid fan.vendor
wait 500
expectmodal 1
clickid modal#1
wait 3500
expectmodal 0
expect fan.cpu.curve $ecBefore
expect fan.gpu.curve $gpuBefore
expect fan.mode auto
expectui fan.d.cpu
shot /tmp/shots/f7_vendor_restored.png
